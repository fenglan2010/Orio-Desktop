多平台数据同步：
class MultiPlatformSyncEngine:
    """多平台数据同步引擎"""
    
    def __init__(self):
        self.adapters = {
            'ctrip': CtripAdapter(),
            'meituan': MeituanAdapter(),
            'fliggy': FliggyAdapter(),
            'qunar': QunarAdapter()
        }
    
    async def sync_all_platforms(self, hotel_id):
        """同步所有平台数据"""
        tasks = []
        
        for platform, adapter in self.adapters.items():
            task = self._sync_single_platform(adapter, hotel_id)
            tasks.append(task)
        
        # 并行执行
        results = await asyncio.gather(*tasks, return_exceptions=True)
        
        # 数据聚合
        aggregated_data = self._aggregate_results(results)
        
        return aggregated_data
    
    async def _sync_single_platform(self, adapter, hotel_id):
        """同步单个平台数据"""
        try:
            # 增量同步策略
            last_sync_time = await self._get_last_sync_time(hotel_id, adapter.platform)
            
            # 获取数据
            performance_data = await adapter.fetch_performance(
                hotel_id, 
                since=last_sync_time
            )
            
            reviews = await adapter.fetch_reviews(hotel_id, limit=100)
            
            competitor_data = await adapter.fetch_competitors(hotel_id)
            
            return {
                'platform': adapter.platform,
                'performance': performance_data,
                'reviews': reviews,
                'competitors': competitor_data,
                'status': 'success'
            }
            
        except Exception as e:
            return {
                'platform': adapter.platform,
                'error': str(e),
                'status': 'failed'
            }
